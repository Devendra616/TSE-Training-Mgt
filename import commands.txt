Convert to UTF-8
iconv -f WINDOWS-1252 -t UTF-8 department.csv > department_utf8.csv

docker cp department_utf8.csv training-mgt-db:/tmp/department.csv
docker cp employees_utf8.csv training-mgt-db:/tmp/employees.csv

CREATE TEMP TABLE departments_import(deptcode integer, name text);

COPY departments_import FROM '/tmp/department.csv' WITH (    FORMAT csv,    HEADER true);

CREATE TEMP TABLE employees_import (    empid text,    sap text,    empname text,    desig text,    grade text,    scategory text,    sex text,    department text,    deptcode integer);

COPY employees_import FROM '/tmp/employees.csv' WITH (    FORMAT csv,    HEADER true);

INSERT INTO departments (    name,    created_at,    updated_at) SELECT    name,    NOW(),    NOW() FROM departments_import;

INSERT INTO employees (    sap_id,    full_name,    designation,    department_id,    status,    created_at,    updated_at) SELECT    sap,    empname,    desig,    deptcode,    'active',    NOW(),    NOW() FROM employees_import;
